Data preprocessing

We merge two dataframes on the date, home_team, and away_team to get the goalscorers associated with each match. We then filter out any matches that do not have a known outcome (i.e., both home_score and away_score are NA).

results_with_scorers <- results |>
  left_join(goalscorers, by = c("date", "home_team", "away_team")) |>
  filter(!is.na(home_score) | !is.na(away_score))

For visual appeal, we add country flags using Unicode characters. We create a function get_flag that takes country codes and returns the corresponding flag emoji. We then join this information to our main dataframe.

# function to get the unicode for the country flag
get_flag <- function(country_codes) {
  sapply(country_codes, function(country_code) {
    # question mark emoji
    if (is.null(country_code) || is.na(country_code)) {
      return(intToUtf8(10067))
    } else {
      intToUtf8(127397 + strtoi(charToRaw(toupper(country_code)), 16L))
    }
  }) |>
    as.vector()
}

# some manual iso codes
manual_iso <- tribble(

country_codes <-
  bind_rows(
    select(countrycode::codelist, c(country.name.en, iso2c, continent)),
    manual_iso

  na.omit()

results_with_scorers <- results_with_scorers |>
  left_join(
    select(country_codes, country.name.en, iso2c),
    by = c("home_team" = "country.name.en")
  ) |>
  mutate(country_flag_home = get_flag(iso2c)) |>
  select(-iso2c) |>
  left_join(
    select(country_codes, country.name.en, iso2c),
    by = c("away_team" = "country.name.en")

  mutate(country_flag_away = get_flag(iso2c)) |>
  select(-iso2c) |>
  left_join(
    country_codes |>
      select(country.name.en, continent),
    by = c("country" = "country.name.en")
  )

It would be nice to have the longitude and latitude of each city where a match was played. In order to achieve this, we will use the cities15000.txt dataset from GeoNames (https://download.geonames.org/export/dump/). This dataset contains all cities with a population of over 15,000 people, along with their geographical coordinates.

There are few challenges to consider when merging the city data with our match data: - City names may not be unique (e.g., there are multiple cities named “Springfield” in the USA). To address this, we will also use the country code to help identify the correct city. - Some city names in our match data may not exactly match the names in the GeoNames dataset (e.g., “Frankfurt” vs. “Frankfurt am Main”). We will handle this by first attempting to match on the exact city name, and then using alternate names from the GeoNames dataset if the initial match fails.

cities <- fread(
  "../data/cities15000.txt",
  sep = "\t",
  header = FALSE,
  fill = TRUE,
  quote = "",
  na.strings = c("", "NA")
)

setnames(
  cities,
  c(
    "geonameid",
    "name",
    "asciiname",
    "alternatenames",
    "latitude",
    "longitude",
    "feature_class",
    "feature_code",
    "country_code",
    "cc2",
    "admin1",
    "admin2",
    "admin3",
    "admin4",
    "population",
    "elevation",
    "dem",
    "timezone",
    "modification_date"
  )
)

# Keep only the most “important” city per name/country
cities_clean <- cities |>
  group_by(name, country_code) |>
  slice_max(population, n = 1, with_ties = FALSE) |>
  ungroup()

# Merge longitude and latitude into data_matches
results_with_scorers <- results_with_scorers |>
  # Change city Frankfurt to Frankfurt am Main to match the city database
  mutate(city = ifelse(city == "Frankfurt", "Frankfurt am Main", city)) |>
  # Add country code so we can match cities better
  left_join(
    country_codes |>
      select(country.name.en, iso2c),
    by = c("country" = "country.name.en")
  ) |>
  rename(country_code = iso2c) |>
  left_join(
    cities_clean |>
      select(name, country_code, latitude, longitude),
    by = c("city" = "name", "country_code" = "country_code")
  )

# If NA's in latitude/longitude, try to match on alternate names
results_with_scorers_na <- results_with_scorers |>
  filter(is.na(latitude) | is.na(longitude)) |>
  # remove latitude and longitude to avoid overwriting
  select(-c(latitude, longitude)) |>
  left_join(
    cities_clean |>
      select(alternatenames, country_code, population, latitude, longitude) |>
      # split alternatenames into multiple rows
      mutate(alternatenames = strsplit(as.character(alternatenames), ",")) |>
      unnest(alternatenames) |>
      filter(!is.na(alternatenames) & alternatenames != "") |>
      # keep only the most populous city per alternate name/country
      # this avoids multiple matches
      # quite "expensive" operation, but managable for a one-off
      group_by(alternatenames, country_code) |>
      slice_max(population, n = 1, with_ties = FALSE) |>
      ungroup(),
    by = c("city" = "alternatenames", "country_code" = "country_code")
  ) |>
  select(names(results_with_scorers))

results_with_scorers <- results_with_scorers |>
  filter(!(is.na(latitude) | is.na(longitude))) |>
  bind_rows(results_with_scorers_na)

We didn’t manage to resolve all NA’s in latitude/longitude, but this should be sufficient for now. In total we have 2204 matches without coordinates, coming from 527 unique cities.

Identifiers

To facilitate filtering and grouping in the Shiny app, we create unique identifiers for matches (match_id). This is a combination of the date, home team, and away team.

results_with_scorers <- results_with_scorers |>
  mutate(
    match_id = paste0(date, "_", home_team, "_vs_", away_team)
  )

Save data for use in Shiny

For maximum compatibility between Python and R we will save the data back again as .csv files, with the downside being that there is no strict preservation of types. Another option would be Parquet (for better type preservation) but that would require additional libraries in Python and R.

write.csv(results_with_scorers, "../data/results_with_scorers.csv")
